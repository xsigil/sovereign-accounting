\begin{equation}
    \Delta \text{Equity}_t \equiv \text{Rev}_t - \text{Exp}_t
\end{equation}
\begin{equation}
    \mathbf{\text{Equity}_t = \text{Equity}_{t-1} + \Delta \text{Equity}_t}
\end{equation}
\begin{equation}
    \text{Equity}_n = \text{Equity}_0 + \sum_{k=1}^n (\text{Rev}_k - \text{Exp}_k)
\end{equation}
